## 💡 DEVELOPMENT GUIDELINES

### Backend Patterns
```javascript
// ALWAYS use ES6 imports
import express from 'express';  // ✅
import axios from 'axios';      // ✅ for Supabase

// Memory-conscious processing
for (const item of items) { // ✅ 
  // process item
}

// API responses always include success flag
res.json({ success: true, data: result });

// Supabase with axios (not native fetch)
const response = await axios.get(`${supabaseUrl}/rest/v1/table`, {
  headers: {
    'apikey': supabaseKey,
    'Authorization': `Bearer ${supabaseKey}`
  }
});

// Lazy-loaded Supabase client (prevents import-time failures)
let supabase = null;
async function getSupabaseClient() {
  if (supabase) return supabase;
  
  if (!process.env.SUPABASE_URL || !process.env.SUPABASE_ANON_KEY) {
    return null; // Graceful fallback
  }
  
  const { createClient } = await import('@supabase/supabase-js');
  supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_ANON_KEY);
  return supabase;
}
```
